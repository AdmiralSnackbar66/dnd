%%
>[!tip] Review the Characters
>Before we do anything else, it helps to spend a few minutes reviewing the player characters.  What are their names? What do they want? What plays into their backgrounds? What do the players of these characters enjoy to do at the table?
>
>You might not even write anything down during this step, but reviewing the characters helps wire them into your mind - and ensures that the rest of your preparation fits around them.
%%

function updateProperties(notePath) {
    // Ensure Templater is installed
    if (!app.plugins.plugins["templater-obsidian"]) {
        alert("Templater plugin is not enabled!");
        return;
    }
    
    // Step 1: Prompt user for new property value
    let userInput = prompt("Enter the new value to append:");

    if (userInput) {
        // Step 2: Call a Templater user function to update frontmatter
        app.plugins.plugins["templater-obsidian"].templater.run_user_function("appendPropertyToNote", {
            file: notePath,
            userInput: userInput
        });
    }
}

dv.table(["Character Name", "Player", "Race", "Class", "Action"],
    dv.pages('"ttrpgs"')
      .filter(p => p.role === "Player")
      .map(p => [
          dv.fileLink(p.file.path), p.player, p.race, p.class,
          `<button onclick="updateProperties('${p.file.path}')">Mark Session Attendance</button>`
      ])
);
